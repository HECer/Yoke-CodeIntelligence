struct Fixture {
    engine: Engine,
    snapshot: Snapshot,
    graph: Graph,
    root: TempDir,
}
fn write(root: &Path, path: &str, text: &str) {
    let file = root.join(path);
    fs::create_dir_all(file.parent().unwrap()).unwrap();
    fs::write(file, text).unwrap();
}
fn owned(files: Vec<(String, String)>, options: CaptureOptions) -> Fixture {
    let root = TempDir::new().unwrap();
    for (path, text) in files {
        write(root.path(), &path, &text);
    }
    let mut engine = Engine::open(root.path()).unwrap();
    if !options.include_ignored_dirs.is_empty() {
        engine.capture_epoch = crate::analysis::AnalyzerEpoch::V9;
    }
    engine.capture_options = options;
    let (snapshot, graph, _, _) = engine.index(&Deadline::new(120_000)).unwrap();
    Fixture {
        engine,
        snapshot,
        graph,
        root,
    }
}
fn fixture(files: &[(&str, &str)]) -> Fixture {
    owned(
        files
            .iter()
            .map(|(p, t)| (p.to_string(), t.to_string()))
            .collect(),
        CaptureOptions::default(),
    )
}
fn two() -> Fixture {
    fixture(&[
        ("package.json", r#"{"workspaces":["packages/*"]}"#),
        ("packages/a/package.json", r#"{"name":"same"}"#),
        ("packages/b/package.json", r#"{"name":"same"}"#),
        ("packages/a/main.rs", "fn alpha(){ payment(); }\n"),
        ("packages/b/main.rs", "fn beta(){ payment(); }\n"),
    ])
}
fn checked<'f, 'd>(
    f: &'f Fixture,
    d: &'d Deadline,
) -> (CheckedRetrievalRun<'f, 'f, 'd>, CheckedScopePartition<'f>) {
    let mut run = begin(&f.engine.store, &f.snapshot, &f.graph, d, 4_000_000).unwrap();
    let partition = match probe(&mut run).unwrap() {
        ProbeOutcome::Ready(p) => p,
        ProbeOutcome::NotApplicable(p) => panic!("expected partition, got {}", p.code()),
        ProbeOutcome::UnsupportedPolicy(p) => panic!("supported fixture, got {}", p.code()),
    };
    (run, partition)
}
fn hard<T>(value: RetrievalResult<T>) -> RetrievalHardError {
    match value {
        Err(e) => e,
        Ok(_) => panic!("expected specific hard error, got success"),
    }
}
fn scope<'p>(p: &'p CheckedScopePartition<'_>, root: &str) -> &'p CheckedScope {
    p.scopes()
        .find(|s| s.root() == root)
        .unwrap_or_else(|| panic!("missing scope {root}"))
}
fn owner(p: &CheckedScopePartition<'_>, path: &str) -> String {
    p.scope(&p.owner(path).unwrap()).unwrap().root().to_string()
}
fn membership(s: &CheckedScope, ecosystem: Ecosystem) -> &WorkspaceMembership {
    s.memberships()
        .find(|m| m.ecosystem() == ecosystem)
        .unwrap()
}
fn unsupported(f: &Fixture, code: &str) {
    let d = Deadline::new(120_000);
    let mut run = begin(&f.engine.store, &f.snapshot, &f.graph, &d, 4_000_000).unwrap();
    let p = match probe(&mut run).unwrap() {
        ProbeOutcome::UnsupportedPolicy(p) => p,
        ProbeOutcome::Ready(_) => panic!("unsupported source admitted"),
        ProbeOutcome::NotApplicable(p) => panic!("unsupported source hidden as {}", p.code()),
    };
    assert_eq!(p.code(), code);
    assert_eq!(p.physical_package_count(), None);
    assert!(p.diagnostics().any(|d| d.code() == code));
}
fn captured(f: &Fixture, path: &str, text: &str) {
    let file = f
        .snapshot
        .files
        .iter()
        .find(|row| row.path == path)
        .unwrap();
    assert_eq!(file.size, text.len());
    assert_eq!(file.hash, hash(text.as_bytes()));
}
fn logical_state(f: &Fixture) -> Vec<(String, Vec<String>)> {
    let mut q = f
        .engine
        .store
        .db
        .prepare("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name")
        .unwrap();
    let names: Vec<String> = q
        .query_map([], |r| r.get(0))
        .unwrap()
        .collect::<rusqlite::Result<_>>()
        .unwrap();
    names
        .into_iter()
        .map(|name| {
            let sql = format!("SELECT * FROM \"{}\"", name.replace('"', "\"\""));
            let mut q = f.engine.store.db.prepare(&sql).unwrap();
            let count = q.column_count();
            let mut rows: Vec<String> = q
                .query_map([], |r| {
                    let values: Vec<rusqlite::types::Value> = (0..count)
                        .map(|i| r.get(i))
                        .collect::<rusqlite::Result<_>>()?;
                    Ok(format!("{values:?}"))
                })
                .unwrap()
                .collect::<rusqlite::Result<_>>()
                .unwrap();
            rows.sort();
            (name, rows)
        })
        .collect()
}

#[test]
fn captured_npm_packages_have_distinct_path_ids_and_deepest_owners() {
    let f = two();
    captured(&f, "packages/a/package.json", r#"{"name":"same"}"#);
    let d = Deadline::new(20_000);
    let (run, p) = checked(&f, &d);
    p.validate_for(&run).unwrap();
    assert!(std::ptr::eq(run.deadline(), &d));
    assert!(std::ptr::eq(run.snapshot(), &f.snapshot));
    assert!(std::ptr::eq(run.graph(), &f.graph));
    assert_eq!(p.physical_package_count(), 2);
    assert_eq!(
        p.scopes().map(|s| s.root()).collect::<Vec<_>>(),
        [".", "packages/a", "packages/b"]
    );
    assert_eq!(scope(&p, ".").kind(), ScopeKind::WorkspaceUnassigned);
    assert_eq!(owner(&p, "packages/a/main.rs"), "packages/a");
    assert_eq!(owner(&p, "package.json"), ".");
    assert_ne!(scope(&p, "packages/a").id(), scope(&p, "packages/b").id());
    assert_eq!(
        scope(&p, "packages/a").id(),
        identity(&(
            "yoke-retrieval-scope-1",
            &f.snapshot.workspace_id,
            identity(&f.snapshot),
            "packages/a"
        ))
    );
}

#[test]
fn p1_mixed_ecosystems_and_nested_virtual_barrier_have_seven_scopes() {
    let f = fixture(&[
        ("package.json", r#"{"private":true}"#),
        (
            "pnpm-workspace.yaml",
            "packages:\n  - 'packages/*'\n  - '!packages/excluded'\n",
        ),
        ("packages/a/package.json", r#"{"name":"same"}"#),
        ("packages/a/Cargo.toml", "[package]\nname='mixed'\n"),
        ("packages/b/package.json", r#"{"name":"same"}"#),
        ("packages/excluded/package.json", r#"{"name":"excluded"}"#),
        (
            "packages/a/nested/pnpm-workspace.yaml",
            "packages:\n  - 'mods/*'\n",
        ),
        ("packages/a/nested/mods/c/package.json", r#"{"name":"c"}"#),
        ("standalone/package.json", r#"{"name":"same"}"#),
        ("src/root.rs", "fn root(){}\n"),
        ("packages/a/src/a.rs", "fn a(){}\n"),
        ("packages/a/nested/README.md", "# Nested workspace\n"),
        ("packages/a/nested/mods/c/src/c.rs", "fn c(){}\n"),
        ("packages/b/src/b.rs", "fn b(){}\n"),
        ("packages/excluded/src/x.rs", "fn x(){}\n"),
        ("standalone/src/s.rs", "fn s(){}\n"),
    ]);
    let d = Deadline::new(20_000);
    let (_, p) = checked(&f, &d);
    assert_eq!(p.physical_package_count(), 5);
    assert_eq!(p.scopes().count(), 7);
    for (path, expected) in [
        ("src/root.rs", "."),
        ("packages/a/src/a.rs", "packages/a"),
        ("packages/a/nested/README.md", "packages/a/nested"),
        (
            "packages/a/nested/mods/c/src/c.rs",
            "packages/a/nested/mods/c",
        ),
        ("packages/b/src/b.rs", "packages/b"),
        ("packages/excluded/src/x.rs", "packages/excluded"),
        ("standalone/src/s.rs", "standalone"),
    ] {
        assert_eq!(owner(&p, path), expected);
    }
    assert_eq!(
        scope(&p, "packages/a/nested").kind(),
        ScopeKind::WorkspaceUnassigned
    );
    assert_eq!(
        scope(&p, "packages/a").labels().count(),
        2,
        "one physical share"
    );
    let c = membership(scope(&p, "packages/a/nested/mods/c"), Ecosystem::Pnpm);
    assert_eq!(c.workspace_root(), Some("packages/a/nested"));
    assert_eq!(c.reason(), MembershipReason::Member);
    assert_eq!(
        membership(scope(&p, "packages/excluded"), Ecosystem::Pnpm).reason(),
        MembershipReason::Excluded
    );
    assert_eq!(
        membership(scope(&p, "standalone"), Ecosystem::Pnpm).reason(),
        MembershipReason::Unmatched
    );
}

#[test]
fn documents_keep_original_entity_ranges_and_overlaps_charge_again() {
    let source = "mod parent { pub fn child(){ payment(); } }\n";
    let f = fixture(&[
        ("a/package.json", r#"{"name":"a"}"#),
        ("b/package.json", r#"{"name":"b"}"#),
        ("a/lib.rs", source),
    ]);
    let d = Deadline::new(20_000);
    let (mut run, _) = checked(&f, &d);
    run.read_source("a/lib.rs").unwrap();
    let parent = f
        .graph
        .entities
        .iter()
        .find(|e| e.path == "a/lib.rs" && e.name == "parent")
        .unwrap();
    let child = f
        .graph
        .entities
        .iter()
        .find(|e| e.path == "a/lib.rs" && e.name == "child")
        .unwrap();
    for e in [parent, child] {
        assert_eq!(run.document_cost(e).unwrap(), e.end_byte - e.start_byte);
        let doc = run.admit_document(e).unwrap();
        assert!(std::ptr::eq(doc.entity(), e));
        assert_eq!(doc.text(), &source[e.start_byte..e.end_byte]);
        assert_eq!(doc.source().content_hash(), hash(source.as_bytes()));
    }
    let used = parent.end_byte - parent.start_byte + child.end_byte - child.start_byte;
    assert_eq!(run.usage().document_bytes_charged(), used);
    run.admit_document(child).unwrap();
    assert_eq!(run.usage().document_bytes_charged(), used);
    let mut forged = child.clone();
    forged.end_byte = source.len() + 1;
    assert!(matches!(
        hard(run.admit_document(&forged)),
        RetrievalHardError::SourceIntegrity(_)
    ));
}

fn padded_json(size: usize) -> String {
    let prefix = r#"{"name":"p","padding":""#;
    let suffix = "\"}";
    let result = format!(
        "{prefix}{}{suffix}",
        "x".repeat(size - prefix.len() - suffix.len())
    );
    assert_eq!(result.len(), size);
    result
}

#[test]
fn expired_and_cancelled_runs_keep_typed_errors_without_new_deadline() {
    let f = two();
    let expired = Deadline::new(0);
    assert!(matches!(
        hard(begin(
            &f.engine.store,
            &f.snapshot,
            &f.graph,
            &expired,
            4_000_000
        )),
        RetrievalHardError::Deadline(_)
    ));
    let parent = Deadline::new(20_000);
    let cancelled = Arc::new(AtomicBool::new(false));
    Deadline::run_bounded(&parent, cancelled.clone(), || {
        let d = Deadline::new(20_000);
        let mut run = begin(&f.engine.store, &f.snapshot, &f.graph, &d, 4_000_000).unwrap();
        assert!(matches!(probe(&mut run).unwrap(), ProbeOutcome::Ready(_)));
        let before = run.usage();
        cancelled.store(true, Ordering::SeqCst);
        assert!(matches!(
            hard(run.read_source("packages/a/main.rs")),
            RetrievalHardError::Cancelled(_)
        ));
        assert!(matches!(
            hard(run.admit_document(&f.graph.entities[0])),
            RetrievalHardError::Cancelled(_)
        ));
        assert_eq!(run.usage(), before);
        assert!(std::ptr::eq(run.deadline(), &d));
        cancelled.store(false, Ordering::SeqCst); // Only after the cancellation oracles.
        Ok(())
    })
    .unwrap();
}
