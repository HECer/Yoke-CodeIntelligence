// Append to the private driver test module only after root promotion release.
#[test]
fn ready_global_filter_fallback_omits_balanced_only_scope_counts_with_exact_total() {
    let directory = tempfile::tempdir().unwrap();
    for (path, text) in [
        ("package.json", "{\"workspaces\":[\"packages/*\"]}"),
        ("packages/a/package.json", "{\"name\":\"a\"}"),
        ("packages/b/package.json", "{\"name\":\"b\"}"),
        ("packages/a/main.rs", "fn payment_a(){ payment(); }\n"),
        ("packages/b/main.rs", "fn payment_b(){ payment(); }\n"),
    ] {
        let file = directory.path().join(path);
        fs::create_dir_all(file.parent().unwrap()).unwrap();
        fs::write(file, text).unwrap();
    }
    let mut engine = Engine::open(directory.path()).unwrap();
    engine.capture_epoch = AnalyzerEpoch::V9;
    let (snapshot, graph, _, _) = engine.index(&Deadline::new(30000)).unwrap();
    let before = state(&engine.store);
    let mut results = Vec::new();
    for (path, eligible, expected_names) in [
        ("packages/a", 1, vec!["payment_a"]),
        ("packages/missing", 0, vec![]),
    ] {
        let args = json!({"query":"payment","relativePath":path,"pathGlob":"**/*.rs","graphWeight":0,"limit":200,"perFileLimit":200,"includeBody":true});
        let global = find_global_legacy(
            &engine.store,
            &snapshot,
            &graph,
            &args,
            &Deadline::new(30000),
        )
        .unwrap();
        let names = global["items"]
            .as_array()
            .unwrap()
            .iter()
            .map(|item| item["name"].as_str().unwrap())
            .collect::<Vec<_>>();
        assert_eq!(
            names, expected_names,
            "The saved-corpus global control must succeed before optional metadata assertions"
        );
        let automatic = find_compatibility_auto(
            &engine.store,
            &snapshot,
            &graph,
            &args,
            &Deadline::new(30000),
            4_000_000,
        )
        .unwrap();
        let scope = &automatic["retrievalScope"];
        assert_eq!(scope["effectiveMode"], "global");
        assert_eq!(scope["fallback"]["kind"], "query_filter");
        assert_eq!(scope["queryEligibleScopeCount"], eligible);
        assert_eq!(scope["actualPackageRootCount"], 2);
        assert!(
            scope["partitionId"]
                .as_str()
                .is_some_and(crate::model::valid_id)
        );
        assert_eq!(scope["admission"]["admittedSourceCount"], eligible);
        assert_eq!(scope["probe"]["markerFilesChecked"], 3);
        let mut without_metadata = automatic.clone();
        without_metadata
            .as_object_mut()
            .unwrap()
            .remove("retrievalScope");
        assert_eq!(
            without_metadata, global,
            "Same-run fallback must retain actual old global arithmetic/items/admission controls"
        );
        results.push(automatic);
    }
    for result in results {
        assert_eq!(
            result["retrievalScope"]["scopes"],
            json!([]),
            "A global result must not expose invented zero balanced-corpus counters"
        );
        assert_eq!(
            result["retrievalScope"]["omittedScopeSummaryCount"], 3,
            "The checked partition has two package scopes plus the root-unassigned scope"
        );
    }
    assert_eq!(state(&engine.store), before);
}
